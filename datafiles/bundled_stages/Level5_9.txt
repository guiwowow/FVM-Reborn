_VM_CARD_CREATED {
	card_id = VM_GetLastCreatedCard()
	card_type = VM_GetProp(card_id,"plant_id")
	if(card_type=="player"){
		row = VM_GetProp(card_id,"grid_row")
		col = VM_GetProp(card_id,"grid_col")
		VM_SwapPlants(col, row, player_id, -1)
		if (col <= 2) {
			VM_SpawnPlant("soda_bubble", col, row, 0, 0, 0)
		}// 把角色移到场景外面
		player_id =player_id +  1 
		exit
	}
	
    plant_id = VM_GetLastCreatedCard()                            // 获取刚种下的植物实例 ID	
    name = VM_GetProp(plant_id, "plant_id")                       // 读取植物卡片名
	set_col = VM_GetProp(plant_id, "grid_col")                    // 读取植物列数
	set_row = VM_GetProp(plant_id, "grid_row")                    // 读取植物行数
	set_cost = VM_GetProp(plant_id, "cost")                       // 读取植物耗火
	
	if (name != "soda_bubble"){                                   // 若不是放置气泡
		lilypad = VM_GetPlantAt(set_col, set_row, "lilypad")      // 获取本格承载卡实例
		if (lilypad == -1){                                       // 若承载卡没有卡片
			VM_ClearPlants(set_col, set_row)                      // 清除本格卡片
			VM_SetCardSlotProp(name, "cooldown_timer", 9999)      // 重置刚种下的植物冷却

			VM_SetFlame(had_flame+ set_cost)                      // 返还火苗
			VM_ShowNotice("请先放气泡")                            // 给予玩家提示	
		}
	}else{
		if(set_row!=0){
			if(set_row!=6){
				VM_SetProp(plant_id,"plant_type" , "lilypad")

	}																		// 每帧计时器-1，至-1为止

_VM_WAVE_START {
	w = VM_GetWave()									// 当前波数
	VM_ShellPrint("当前大波数:",w)
	sw = w * 10											// 修正小波数
	enemy_cost = VM_Ceil(sw / 3.0)						// 每3波小波出怪量+1，向上取整
	if(w > 0){
		attendant_count = enemy_cost					// 计算旗帜随从数
		enemy_cost = VM_Floor(enemy_cost * 2.5)			// 旗帜波2.5倍出怪量
		random_spawn = VM_Random(1, 5)
		VM_SpawnEnemy("lambo_mouse", random_spawn, 100)
		enemy_cost = enemy_cost - 1						// 生成旗帜并占用对应出怪量
		if (attendant_count > 8){
			attendant_count = 8
		}												// 限制随从数不超过8
		while(attendant_count > 0){
			random_spawn = VM_Random(1, 5)
			VM_SpawnEnemy("normal_mouse", random_spawn, 100)
			enemy_cost = enemy_cost - 1
			attendant_count = attendant_count - 1
		}												// 生成随从平民鼠并占用对应出怪量
	}
	if (sw < 5) {
		enemy_class = 1									// 第1波解锁1阶段僵尸(普僵、旗帜、路障、铁桶、读报)
	}elif(sw < 10){
		enemy_class = 2									// 第5波解锁2阶段僵尸(撑杆、铁门、橄榄、舞王)
	}elif(sw < 15){
		enemy_class = 3									// 第10波解锁3阶段僵尸(潜水、冰车、雪橇、海豚、小丑、气球、矿工、跳跳、蹦极、扶梯、投篮)
	}else{
		enemy_class = 4									// 第15波解锁4阶段僵尸(巨人)
	}
	while(enemy_cost > 0){									// 循环抽怪至出怪量耗尽
		random_spawn = VM_Random(1, 5)						// 随机行数
		if (enemy_class <= 2){								// 1、2阶段
			if (enemy_cost == 1) {			
				random_enemy = VM_Random(1, 4000)
			}elif (enemy_cost < 4) {
				random_enemy = VM_Random(1, 8000)
			}else{
				random_enemy = VM_Random(1, 12000)
			}
			if( random_enemy <= 4000) {
				VM_SpawnEnemy("normal_mouse", random_spawn, 100)				//普通僵尸：权重4000，等级1
				enemy_cost = enemy_cost - 1
			}elif (random_enemy <= 8000) {
				VM_SpawnEnemy("football_fan_mouse", random_spawn, 285)			//路障僵尸：权重4000，等级2
				football_fan = VM_GetLastCreatedEnemy()
				VM_SetProp(football_fan, "helmet_max_hp", 185)
				VM_SetProp(football_fan, "helmet_hp", 185)
				enemy_cost = enemy_cost - 2
			}else {
				VM_SpawnEnemy("iron_pan_mouse", random_spawn, 650)				//铁桶僵尸：权重4000，等级4
				iron_pan = VM_GetLastCreatedEnemy()
				VM_SetProp(iron_pan, "helmet_max_hp", 550)
				VM_SetProp(iron_pan, "helmet_hp", 550)
				enemy_cost = enemy_cost - 4
			}
		}
		if (enemy_class == 3) {													//3阶段
			if (enemy_cost == 1) {			
				random_enemy = VM_Random(1, 4000)
			}elif (enemy_cost == 2) {

			}elif (random_enemy <= 8000) {
