// ============================================================
//  LabMap 脚本完整模板
//
//  使用方法：
//  1. 所有事件块（_VM_*）都已就位，块内语句默认全部注释
//  2. 删除语句行首的 // 即可启用该语句
//  3. 事件块只有到对应时机才会自动执行（见各块上方注释）
//
//  语法速查：
//  - 赋值/运算：x = 10    y = x + 5    ok = (x > y)
//    支持 + - * / %，比较 == != > >= < <=
//  - 条件：if (a == 1) { ... } elif (a == 2) { ... } else { ... }
//    （else if 与 elif 等价；elif/else 必须紧跟刚闭合的 if）
//  - 注释：// 单行注释（字符串内的 // 不算注释）
//  - 结束当前块：halt（与 exit 同义）
//  - 一行一句；分号 ; 视作换行；表达式不能跨行
//  - 跨块变量共享：不同块中同名变量共用同一个内存槽，可跨事件传值
//
//  提示：打开"帮助"按钮可查看完整语法手册
// ============================================================

_VM_CARD_DESTROYED {
    dead_plant = VM_GetLastDestroyedCard()      // 刚销毁的植物实例 ID
	dead_count = VM_ArraySize("small_fire")
	dead_plant_id = VM_GetKilledProp("plant_id")
	if (dead_plant_id == "small_fire") {
		while (dead_count > 0) {
			fire_list = VM_ArrayGet("small_fire",dead_count)
			if (fire_list == dead_plant_id){
				VM_ArrayDel("small_fire",dead_count)
				VM_ArrayDel("small_fire_hp",dead_count)
			}
			dead_count = dead_count - 1
		}
	}

	if (start == 0){
		start = 1
	}

		if (VM_GetEnemyCount() == 0){
			VM_GameLose()
		}
	}

		no_flame = 1

		had_small_fire = VM_ArrayGet("small_fire",small_fire_count)

		        f = VM_CreateInstance("flame", VM_GetProp(had_small_fire, "x"), VM_GetProp(had_small_fire, "y") - 60)

		        VM_SetProp(f, "value", 25)
			}
		}
		VM_ArraySet("small_fire_hp",small_fire_count,hp_now)
		small_fire_count = small_fire_count - 1
	}
