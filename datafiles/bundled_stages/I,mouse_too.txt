_VM_CARD_CREATED {

		plant_col = VM_GetProp(plant_id, "grid_col")
		plant_row = VM_GetProp(plant_id, "grid_row")
		plant_x = 107 * plant_col + 748.5
		if (plant_name == "xiao_long_bao"){
			VM_ClearPlants(plant_col,plant_row)
			VM_SpawnEnemy("normal_mouse",plant_row,100)
			spawn = VM_GetLastCreatedEnemy()
			VM_ArrayADD("alive", spawn)

			VM_ClearPlants(plant_col,plant_row)

_VM_CARD_DESTROYED {
    dead_plant = VM_GetLastDestroyedCard()      // 刚销毁的植物实例 ID
	small_fire_counter = VM_ArraySize("small_fire")
	alive_brains = VM_ArraySize("brains")
	dead_plant_id = VM_GetKilledProp("plant_id")
	if (dead_plant_id == "small_fire") {
		while (small_fire_counter > 0) {
			fire_list = VM_ArrayGet("small_fire",small_fire_counter)
			if (fire_list == dead_plant){
				VM_ArrayDel("small_fire",small_fire_counter)
				VM_ArrayDel("small_fire_hp",small_fire_counter)
			}
			small_fire_counter = small_fire_counter - 1
		}
	}
	if (dead_plant_id == "cotton_candy") {
		while (alive_brains > 0) {
			brain_list = VM_ArrayGet("brains",alive_brains)
			if (brain_list == dead_plant){
				if(brain_list != 0){
					VM_ArrayDel("brains",alive_brains)
					}
			}
			alive_brains = alive_brains - 1
		}
	}
}
